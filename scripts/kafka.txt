docker run -d \
  --name atlas-kafka \
  -p 9092:9092 \
  apache/kafka:4.1.2

docker exec -it atlas-kafka /opt/kafka/bin/kafka-topics.sh \
  --bootstrap-server localhost:9092 \
  --create \
  --if-not-exists \
  --topic order-events \
  --partitions 3 \
  --replication-factor 1
